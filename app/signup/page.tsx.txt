'use client'
import { useState } from 'react'
import { createClient } from '@/utils/supabase/client'

export default function SignupPage() {
  const [form, setForm] = useState({ full_name: '', email: '' })
  const [sent, setSent] = useState(false)
  const supabase = createClient()

  const handleSignup = async (e: React.FormEvent) => {
    e.preventDefault()

    const { error } = await supabase.auth.signInWithOtp({
      email: form.email,
      options: {
        emailRedirectTo: `${window.location.origin}/auth/verify`,
        data: {
          full_name: form.full_name,
          role: 'employee', // Employee hi signup kar sakta hai
        },
      },
    })

    if (error) alert(error.message)
    else setSent(true)
  }

  if (sent) return (
    <div className="min-h-screen flex items-center justify-center">
      <div className="text-center">
        <h2 className="text-2xl font-bold">✅ Verify karo!</h2>
        <p className="text-gray-500 mt-2">
          {form.email} pe magic link bheja gaya hai.
        </p>
      </div>
    </div>
  )

  return (
    <div className="min-h-screen flex items-center justify-center bg-gray-50">
      <div className="bg-white p-8 rounded-xl shadow-md w-full max-w-md">
        <h1 className="text-2xl font-bold mb-2">Employee Sign Up</h1>

        <form onSubmit={handleSignup} className="space-y-4">
          <input
            type="text"
            placeholder="Poora naam"
            value={form.full_name}
            onChange={(e) => setForm({ ...form, full_name: e.target.value })}
            required
            className="w-full border rounded-lg px-4 py-2"
          />
          <input
            type="email"
            placeholder="Company email"
            value={form.email}
            onChange={(e) => setForm({ ...form, email: e.target.value })}
            required
            className="w-full border rounded-lg px-4 py-2"
          />
          <button
            type="submit"
            className="w-full bg-green-600 text-white py-2 rounded-lg hover:bg-green-700"
          >
            Sign Up
          </button>
        </form>
      </div>
    </div>
  )
}
